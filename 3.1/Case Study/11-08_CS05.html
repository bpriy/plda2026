### 2. Use GeCo for corruption (trial-and-error) ###
# https://dmm.anu.edu.au/geco/ 
# maximum modification per attribute : 4
# number of modifications per record : 4
# zip code, month, day, year - attribute probability of 0.25 for each
# for each, keyboard edit w/ row probability of 1 
# sex - no corruption
# number of duplicates per record is 1 
# i.e., corrupts record but also keeps original record in the data
# for duplicated records, we remove the original record below
# trial-and-error of number of duplicate records (# records corrupted)
#' to get corrupted matching variables data that would result 
#' in roughly the desired mismatch rates if linked 
#' to the uncorrupted, original data (srsfakemv)

# function used to check whether true mismatch rate would be as desired
checkmismatchrate_fun <- function(srsfakemvcorrupt){
  # use fastLink() to link dfA and dfB
  flresults <- fastLink(dfA = srsfakemv, dfB = srsfakemvcorrupt, 
                        varnames = c("sex", "zipcode", "year", "month", "day"),
                        numeric.match = c("sex", "zipcode", "year", "month", "day"),
                        return.all = FALSE, dedupe.matches = FALSE,
                        return.df = TRUE, threshold.match = 0.0000000000000001)
  
  # convert to matrix form for linear assignment problem
  # https://stackoverflow.com/questions/47037504/r-converting-a-data-frame-of-row-column-indices-to-a-matrix
  flresults <- as.data.frame(cbind(dfa = flresults$matches$inds.a, dfb = flresults$matches$inds.b, post.prob = flresults$posterior))
  ppmatrix <- matrix(0, 1151, 1151)
  ppmatrix[as.matrix(flresults[1:2])] <- flresults$post.prob
  
  # obtain one-to-one matches
  matches <- as.data.frame(cbind(dfa = c(1:1151), dfb = solve_LSAP(ppmatrix, maximum = TRUE)))
  matches$match <- ifelse(matches$dfa == matches$dfb, 1, 0)
  matches <- left_join(matches, flresults, by = c("dfa", "dfb"))
  
  # mismatch rate
  return(list(matches = matches, alpha = 1-sum(matches$match)/1151))
}

## 1. Unconstrained Adjustment Method ##

fit_mixture_Cox <- function(X, y, cens, mp, logisRegMp, link_function, is_flagged, logitbound){
  #### input variables
  X <- as.matrix(X)
  n <- nrow(X)
  p <- ncol(X)
  y <- as.numeric(y)
  
  ##### if missing inputs
  if(missing(mp)){
    mp <- matrix(nrow = n, ncol = 1, data = 1)
  }
  if(missing(logisRegMp)){
    logisRegMp = TRUE
  }
  if(missing(link_function)){
    link_function = "identity"
  }
  if(missing(is_flagged)){
    is_flagged = rep(FALSE,n)
  }
  
  ##### custom error message
  if(length(y) != n)
    stop("Error: Length of 'y' must be equal to the number of rows of
      'X' \n")
  
  #### initial baseline estimator for corrected matched data
  creg <- coxph(Surv(y, event = 1 - cens) ~ X - 1)
  beta_cur <- creg$coef
  mu <- X %*% creg$coef
  
  Breslow_Estimator <- basehaz(creg, centered=FALSE)
  cumhazard <- Breslow_Estimator$hazard
  times <- Breslow_Estimator$time
  #' using option 2 of defining hazard at t_1: \hat{\Lambda}_0(t_{(1)}) for now
  D1 <-c(cumhazard[1], diff(cumhazard)/diff(times)) # (!) updated
  
  lambdahat_0 <- D1[match(y,times)]
  Lambdahat_0 <- cumhazard[match(y,times)]
  
  #### initial estimator for baseline for mismatched data
  survf0 <- survfit(Surv(y, event = 1 - cens) ~ 1)
  times <- survf0$time
  #' using option 2 of defining hazard at t_1: \hat{\Lambda}_0(t_{(1)}) for now
  D <- c(survf0$cumhaz[1], diff(survf0$cumhaz)/diff(times)) # (!) updated
  
  g_lambdahat_0 <- D[match(y,times)] # (!) updated
  g_Lambdahat_0 <- survf0$cumhaz[match(y,times)] # (!) updated
  
  #### fymu and fy
  fymu <- function(mu, cens, l0, L0) (exp(mu) * l0)^(1 - cens) *
    exp(-exp(mu) * L0)
  fy <- g_lambdahat_0^(1-cens) * exp(-g_Lambdahat_0)
  
  #### nloglik w/ modifications
  nloglik <- function(mu, cens, hs, l0, L0) sum(-log(hs[!is_flagged] *
                                                       fymu(mu, cens, l0, L0)[!is_flagged] + (1 - hs[!is_flagged])*
                                                       fy[!is_flagged])) - sum(log(fymu(mu, cens, l0, L0)[is_flagged]))
  
  #### logistic model for mismatches
  hgamma_gamma <- function(gamma, subs){
    return(list(fun = rep(gamma, sum(subs)), dfun = rep(1, sum(subs)), d2fun = rep(0, sum(subs))))
  }
  
  hs <- hgamma_gamma(0.5, !is_flagged)$fun
  
  if(logisRegMp){
    if(link_function == "identity"){
      logit_ps <- mp
    }
    if(link_function == "logit"){
      logit_ps <- log(mp / (1 - mp))
    }
    Delta <- cbind(logit_ps)
    
    hgamma <- function(eta){
      pi <- plogis(eta)
      return(list(fun = pi, dfun = pi*(1-pi), d2fun = pi*(1-pi)*(1 - 2*pi)))
    }
    
    gammacur <- rep(-logitbound, ncol(Delta))
    hs <- hgamma(Delta %*% gammacur)$fun
  }
  
  #### while loop initial variables
  iter <- 1
  maxiter <- 1000
  tol <- 1E-4
  objs <- numeric(maxiter)
  nloglik_cur <- nloglik(mu, cens, hs, lambdahat_0, Lambdahat_0)
  objs[iter] <- nloglik_cur
  pcur = rep(0,n)
  
  track_beta <- matrix(0, nrow = maxiter, ncol = p)
  track_lam <- matrix(0, nrow = maxiter, ncol = n)
  track_Lam <- matrix(0, nrow = maxiter, ncol = n)
  
  track_beta[1,] <- beta_cur
  track_lam[1,] <- lambdahat_0
  track_Lam[1,] <- Lambdahat_0
  
  lambdahat_0_ <- lambdahat_0
  Lambdahat_0_ <- Lambdahat_0
  
  #### while loop - EM algorithm
  while(iter < maxiter){
    num <- hs[!is_flagged] * fymu(mu, cens, lambdahat_0_,
                                  Lambdahat_0_)[!is_flagged]
    denom <- num + (1-hs[!is_flagged]) * fy[!is_flagged]
    pcur[!is_flagged] <- num/denom
    pcur[is_flagged] <- 1
    
    #gammacur <- mean(pcur)#plogis(log(4))
    #hs <- hgamma_gamma(mean(pcur), !is_flagged)$fun
    #hs <- hgamma_gamma(plogis(log(4)), !is_flagged)$fun
    
    if(logisRegMp){
      glm_h <- glm(pcur[!is_flagged] ~ Delta[!is_flagged,] - 1, family = quasibinomial)
      gammacur <- max(coef(glm_h), -logitbound)
      hs <- hgamma(Delta %*% gammacur)$fun
    }
    
    creg <- coxph(Surv(y, event = 1 - cens) ~ X - 1, weights =
                    pmax(drop(pcur),1E-6))
    mu <- X %*% creg$coefficients
    beta_cur <- creg$coefficients
    track_beta[iter+1,] <- beta_cur
    
    Breslow_Estimator <- basehaz(creg, centered=FALSE)
    cumhazard <- Breslow_Estimator$hazard
    times <- Breslow_Estimator$time
    #' for now, using option 2 of defining hazard at t_1:
    #' \hat{\Lambda}_0(t_{(1)})
    D1 <- c(cumhazard[1], diff(cumhazard)/diff(times)) # (!) updated
    
    lambdahat_0_ <- D1[match(y,times)]
    Lambdahat_0_ <- cumhazard[match(y,times)]
    
    track_lam[iter+1,] <- lambdahat_0_
    track_Lam[iter+1,] <- Lambdahat_0_
    iter <- iter + 1
    
    objs[iter] <- nloglik(mu, cens, hs, lambdahat_0_, Lambdahat_0_)
    if(objs[iter] + tol > objs[iter-1]){
      break
    }
  }
  
  #### return output w/ modifications
  return(list(betahat =  beta_cur, gammahat = gammacur,
              hshatoriginal = hs, hshatndonly = hs[!is_flagged],
              objective = objs, track_beta = track_beta,
              track_Lam = track_Lam, track_lam = track_lam,
              Lambdahat_0 = Lambdahat_0_,
              g_Lambdahat_0= g_Lambdahat_0, iter=iter, creg = creg))
  
}

constrained_logistic_regression <- function(X, y, bound){
  #browser()
  X <- as.matrix(X)
  d <- ncol(X)
  #log(mean(mu)/(1-mean(mu)))
  #mean(X %*% beta)
  barx <- colMeans(X)


  nloglik_logistic <- function(mu){

      #mu <- plogis(X %*% beta)
      cutoff <- 1E-10 # 1E-8
      flag0 <- (mu < cutoff)
      flag1 <- (mu > 1 - cutoff)
      flag <- flag0 | flag1

      nloglik_nonflag <- sum(-(y[!flag] * log(mu[!flag]) + (1-y[!flag]) * log(1 - mu[!flag])))

      if(any(flag0)){
      if(max(y[flag0]) > cutoff){
        warning("Fitted probabilites zero with non-matching reponse; evaluation of the logistic likelihood for the mismatch indicator may be inaccurate \n")
      }
      }

      if(any(flag1)){
          if(min(y[flag1]) < 1 - cutoff){
               warning("Fitted probabilites one with non-matching reponse; evaluation of the logistic likelihood for the mismatch indicator may be inaccurate \n")
          }
      }
      return(nloglik_nonflag)
  }
  grad_Hess <- function(mu){

    grad <- -crossprod(X, y - mu)
    w <- sqrt(mu * (1-mu))
    Xw <- sweep(X, MARGIN = 1, STATS = w, FUN = "*")
    Hess <- crossprod(Xw)

    return(list(grad = grad, Hess = Hess))
  }

  #betaglm <- coef(glm(y ~ X - 1, family = binomial))
  #betaglm[1] <- betaglm[1] - .45
  betacur <- c(bound, rep(0, d-1))
  mu <- plogis(X %*% betacur)
  iter <- 1
  maxiter <- 1000
  mu <- plogis(X %*% betacur)
  tol <- 1E-8
  objs <- numeric(maxiter)
  objs[1] <- nloglik_logistic(mu)
  #browser()
  while(iter < maxiter){

    gH <- grad_Hess(mu)
    gr <- gH$grad
    H <- gH$Hess

    rhs <- -gr + (H %*% betacur)
    betanew <- solve(H, rhs)

    eta_tmp <- X%*% betanew

    if(mean(eta_tmp) > bound){
      #    betacur <- betanew

      #}
      #else{

      betanew_theta <- solve(rbind(cbind(H,barx), c(barx,0)), c(rhs, bound))
      betanew <- betanew_theta[1:d]
      theta <- betanew_theta[d+1]
    }
    # linesearch
    foo <- function(gamma){
      betagamma <- (1-gamma) * betacur + gamma * betanew
      mu <- plogis(X %*% betagamma)
      nloglik_logistic(mu)
    }
    #gammastar <- optimize(foo, interval = c(0,1))$minimum
    m <- 0
    repeat{
      if(is(try(optimize(foo, interval = c(0,2^-m))$minimum, silent = TRUE),"try-error")){
        m <- m + 1
      } else {
          gammastar <- optimize(foo, interval = c(0,2^-m))$minimum
          break}
      if(2^(-m) < tol) {
        gammastar <- optimize(foo, interval = c(0,2^-m))$minimum
        break}
    }
    betacur <- (1-gammastar) * betacur + gammastar * drop(betanew)
    #if(is.matrix(betacur)){
    #    browser()
    #}
    mu <- plogis(X %*% betacur)
    iter <- iter+1
    objs[iter] <- nloglik_logistic(mu)

    if(iter >= 2){
      if(is.na(objs[iter])) browser()
      if(objs[iter-1] - objs[iter] < tol)
        break
    }

  }

  return(list(beta = betacur, objs = objs))

}

fit_mixture_Cox_constrained <- function(X, y, cens, mp, is_flagged, logitbound, logisRegMp, link_function){
  #### input variables
  X <- as.matrix(X)
  n <- nrow(X)
  p <- ncol(X)
  y <- as.numeric(y)

  ##### if missing inputs
  if(missing(mp)){
    mp = rep(0,n)
  }
  if(missing(logisRegMp)){
    logisRegMp = TRUE
  }
  if(missing(link_function)){
    link_function = "identity"
  }
  if(missing(is_flagged)){
    is_flagged = rep(FALSE,n)
  }

  ##### custom error message
  if(length(y) != n)
    stop("Error: Length of 'y' must be equal to the number of rows of
      'X' \n")

  #### initial baseline estimator for corrected matched data
  creg <- coxph(Surv(y, event = 1 - cens) ~ X - 1)
  beta_cur <- creg$coef
  mu <- X %*% creg$coef

  Breslow_Estimator <- basehaz(creg, centered=FALSE)
  cumhazard <- Breslow_Estimator$hazard
  times <- Breslow_Estimator$time
  #' using option 2 of defining hazard at t_1: \hat{\Lambda}_0(t_{(1)}) for now
  D1 <-c(cumhazard[1], diff(cumhazard)/diff(times)) # (!) updated

  lambdahat_0 <- D1[match(y,times)]
  Lambdahat_0 <- cumhazard[match(y,times)]

  #### initial estimator for baseline for mismatched data
  survf0 <- survfit(Surv(y, event = 1 - cens) ~ 1)
  times <- survf0$time
  #' using option 2 of defining hazard at t_1: \hat{\Lambda}_0(t_{(1)}) for now
  D <- c(survf0$cumhaz[1], diff(survf0$cumhaz)/diff(times)) # (!) updated

  g_lambdahat_0 <- D[match(y,times)] # (!) updated
  g_Lambdahat_0 <- survf0$cumhaz[match(y,times)] # (!) updated

  #### fymu and fy
  fymu <- function(mu, cens, l0, L0) (exp(mu) * l0)^(1 - cens) *
    exp(-exp(mu) * L0)
  fy <- g_lambdahat_0^(1-cens) * exp(-g_Lambdahat_0)

  #### nloglik w/ modifications
  nloglik <- function(mu, cens, hs, l0, L0) sum(-log(hs[!is_flagged] *
                                                       fymu(mu, cens, l0, L0)[!is_flagged] + (1 - hs[!is_flagged])*
                                                       fy[!is_flagged])) - sum(log(fymu(mu, cens, l0, L0)[is_flagged]))

  #### logistic model for mismatches
  hgamma_gamma <- function(gamma, subs){
    return(list(fun = rep(gamma, sum(subs)), dfun = rep(1, sum(subs)), d2fun = rep(0, sum(subs))))
  }

  hs <- hgamma_gamma(0.5, !is_flagged)$fun

  if(logisRegMp){
    if(link_function == "identity"){
      logit_ps <- mp
    }
    if(link_function == "logit"){
      logit_ps <- log(mp / (1 - mp))
    }
    Delta <- cbind(logit_ps)

    hgamma <- function(eta){
      pi <- plogis(eta)
      return(list(fun = pi, dfun = pi*(1-pi), d2fun = pi*(1-pi)*(1 - 2*pi)))
    }

    gammacur <- c(min(logitbound, 0), rep(0, ncol(Delta)-1)) # modified for constrained version
    hs <- hgamma(Delta %*% gammacur)$fun
  }

  #### while loop initial variables
  iter <- 1
  maxiter <- 1000
  tol <- 1E-4
  objs <- numeric(maxiter)
  nloglik_cur <- nloglik(mu, cens, hs, lambdahat_0, Lambdahat_0)
  objs[iter] <- nloglik_cur
  pcur = rep(0,n)

  track_beta <- matrix(0, nrow = maxiter, ncol = p)
  track_lam <- matrix(0, nrow = maxiter, ncol = n)
  track_Lam <- matrix(0, nrow = maxiter, ncol = n)

  track_beta[1,] <- beta_cur
  track_lam[1,] <- lambdahat_0
  track_Lam[1,] <- Lambdahat_0

  lambdahat_0_ <- lambdahat_0
  Lambdahat_0_ <- Lambdahat_0

  #### while loop - EM algorithm
  while(iter < maxiter){
    num <- hs[!is_flagged] * fymu(mu, cens, lambdahat_0_,
                                  Lambdahat_0_)[!is_flagged]
    denom <- num + (1-hs[!is_flagged]) * fy[!is_flagged]
    pcur[!is_flagged] <- num/denom
    pcur[is_flagged] <- 1

    gammacur <- mean(pcur)
    hs <- hgamma_gamma(mean(pcur), !is_flagged)$fun

    if(logisRegMp){  # modified for constrained version
      glm_h <- constrained_logistic_regression(Delta[!is_flagged,], 1-
                                                 pcur[!is_flagged],
                                               logitbound)
      gammacur <- -glm_h$beta
      hs <- hgamma(Delta %*% gammacur)$fun
    }

    creg <- coxph(Surv(y, event = 1 - cens) ~ X - 1, weights =
                    pmax(drop(pcur),1E-6))
    mu <- X %*% creg$coefficients
    beta_cur <- creg$coefficients
    track_beta[iter+1,] <- beta_cur

    Breslow_Estimator <- basehaz(creg, centered=FALSE)
    cumhazard <- Breslow_Estimator$hazard
    times <- Breslow_Estimator$time
    #' for now, using option 2 of defining hazard at t_1:
    #' \hat{\Lambda}_0(t_{(1)})
    D1 <- c(cumhazard[1], diff(cumhazard)/diff(times)) # (!) updated

    lambdahat_0_ <- D1[match(y,times)]
    Lambdahat_0_ <- cumhazard[match(y,times)]

    track_lam[iter+1,] <- lambdahat_0_
    track_Lam[iter+1,] <- Lambdahat_0_
    iter <- iter + 1

    objs[iter] <- nloglik(mu, cens, hs, lambdahat_0_, Lambdahat_0_)
    if(objs[iter] + tol > objs[iter-1]){
      break
    }
  }

  #### return output w/ modifications
  return(list(betahat =  beta_cur, gammahat = gammacur,
              hshatoriginal = hs, hshatndonly = hs[!is_flagged],
              objective = objs, track_beta = track_beta,
              track_Lam = track_Lam, track_lam = track_lam,
              Lambdahat_0 = Lambdahat_0_,
              g_Lambdahat_0= g_Lambdahat_0, iter=iter, creg = creg))

}
